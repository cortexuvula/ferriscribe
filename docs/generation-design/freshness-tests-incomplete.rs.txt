//! Metadata-only input freshness. Never returns or logs clinical content.

#[cfg(test)]
mod tests {
    use super::*;
    use medical_db::generations::{GenerationInsert, GenerationsRepo};
    use medical_db::recordings::RecordingsRepo;
    use medical_core::types::recording::Recording;
    use serde_json::json;

    fn setup() -> (medical_db::Database, uuid::Uuid) {
        let db = medical_db::Database::open_in_memory().unwrap();
        let mut rec = Recording::new("synthetic.wav".into(), "/synthetic/not-read.wav".into());
        rec.transcript = Some("Synthetic transcript".into());
        rec.soap_note = Some("Synthetic SOAP".into());
        rec.letter = Some("Synthetic letter".into());
        let conn = db.conn().unwrap();
        RecordingsRepo::insert(&conn, &rec).unwrap();
        let row = GenerationsRepo::record_generation(&conn, GenerationInsert {
            recording_id: rec.id, output_type: "soap", ai_provider: "local", ai_model: "synthetic",
            prompt_template_name: None, input_transcript: "Synthetic transcript",
            input_context_json: Some(r#"{"context":"Synthetic alpha","patient_context":null}"#),
            draft_text: "Synthetic SOAP",
        }).unwrap();
        GenerationsRepo::update_final_text(&conn, rec.id, "soap", "Synthetic SOAP").unwrap();
        // Only an explicit output→generation binding proves that a capture row
        // belongs to this output. Legacy capture never wrote this binding.
        conn.execute("UPDATE recordings SET metadata = ?1 WHERE id = ?2", rusqlite::params![
            json!({"generation_input_ids": {"soap": row.id.to_string()}}).to_string(), rec.id.to_string()
        ]).unwrap();
        (db, rec.id)
    }

    #[test]
    fn freshness_context_partial_edit_reversion_and_unchanged() {
        let (db, id) = setup();
        let conn = db.conn().unwrap();
        let original = json!({"context":"Synthetic alpha", "patient_context":null});
        let edited = json!({"context":"Synthetic beta", "patient_context":null});
        assert_eq!(read_freshness(&conn, id, &original).unwrap().soap.stale, Some(false));
        assert_eq!(read_freshness(&conn, id, &edited).unwrap().soap.stale, Some(true));
        assert_eq!(read_freshness(&conn, id, &original).unwrap().soap.stale, Some(false));
    }

    #[test]
    fn freshness_transcript_only_and_document_isolation() {
        let (db, id) = setup();
        let conn = db.conn().unwrap();
        conn.execute("UPDATE recordings SET transcript = 'Edited synthetic transcript' WHERE id = ?1", [id.to_string()]).unwrap();
        let result = read_freshness(&conn, id, &json!({"context":"Synthetic alpha", "patient_context":null})).unwrap();
        assert_eq!(result.soap.stale, Some(true));
        assert_eq!(result.letter.stale, None);
        assert_eq!(result.referral.stale, None);
        assert_eq!(result.peer_discussion.stale, None);
    }

    #[test]
    fn freshness_legacy_capture_without_output_binding_is_unknown() {
        let (db, id) = setup();
        let conn = db.conn().unwrap();
        conn.execute("UPDATE recordings SET metadata = '{}' WHERE id = ?1", [id.to_string()]).unwrap();
        let result = read_freshness(&conn, id, &json!({"context":"Synthetic alpha", "patient_context":null})).unwrap();
        assert_eq!(result.soap.stale, None);
        assert_eq!(result.soap.basis, "missing_provenance");
    }

    #[test]
    fn freshness_missing_or_mismatched_output_and_corrupt_context_are_unknown() {
        let (db, id) = setup();
        let conn = db.conn().unwrap();
        conn.execute("UPDATE generations SET input_context_json = 'not-json'", []).unwrap();
        let input = json!({"context":"Synthetic alpha", "patient_context":null});
        assert_eq!(read_freshness(&conn, id, &input).unwrap().soap.stale, None);
        conn.execute("UPDATE recordings SET soap_note = 'Unbound synthetic output' WHERE id = ?1", [id.to_string()]).unwrap();
        assert_eq!(read_freshness(&conn, id, &input).unwrap().soap.stale, None);
    }

    #[test]
    fn freshness_switch_away_back_and_wire_payload_are_metadata_only() {
        let (db, id) = setup();
        let conn = db.conn().unwrap();
        let other = Recording::new("other.wav".into(), "/synthetic/not-read.wav".into());
        RecordingsRepo::insert(&conn, &other).unwrap();
        let input = json!({"context":"Synthetic alpha", "patient_context":null});
        assert_eq!(read_freshness(&conn, other.id, &input).unwrap().soap.stale, None);
        let result = read_freshness(&conn, id, &input).unwrap();
        assert_eq!(result.soap.stale, Some(false));
        let wire = serde_json::to_value(result).unwrap();
        assert_eq!(wire["soap"], json!({"stale":false,"basis":"captured_inputs"}));
        for (_, verdict) in wire.as_object().unwrap() {
            assert_eq!(verdict.as_object().unwrap().len(), 2);
        }
    }
}
